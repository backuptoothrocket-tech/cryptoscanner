import React, { useState } from "react";
import { LogEntry } from "../types";
import { Database, Eye, Trash2, ShieldCheck, AlertTriangle, HelpCircle, CheckCircle2, ChevronDown, ChevronUp, Copy, Check, MessageSquare } from "lucide-react";

interface LogViewerProps {
  logs: LogEntry[];
  onClear: () => void;
}

export default function LogViewer({ logs, onClear }: LogViewerProps) {
  const [selectedLogId, setSelectedLogId] = useState<string | null>(null);
  const [copiedId, setCopiedId] = useState<string | null>(null);

  const toggleDetails = (id: string) => {
    setSelectedLogId(prev => (prev === id ? null : id));
  };

  const copyAlertToClipboard = (log: LogEntry, text: string) => {
    navigator.clipboard.writeText(text);
    setCopiedId(log.id);
    setTimeout(() => setCopiedId(null), 2000);
  };

  return (
    <div className="bg-slate-900 border border-slate-800 rounded-xl overflow-hidden shadow-lg space-y-4 p-5" id="log-viewer">
      {/* Logger Header */}
      <div className="flex items-center justify-between border-b border-slate-800 pb-3">
        <div className="flex items-center gap-2">
          <Database className="w-5 h-5 text-cyan-400" />
          <h3 className="text-sm font-semibold text-slate-100 uppercase tracking-wider">Evaluation Pipeline Logs</h3>
        </div>
        
        {logs.length > 0 && (
          <button
            onClick={onClear}
            className="flex items-center gap-1 text-slate-400 hover:text-rose-400 text-xs px-2.5 py-1 rounded bg-slate-950/80 hover:bg-rose-950/20 border border-slate-800 hover:border-rose-500/20 active:scale-95 transition-all cursor-pointer"
            id="clear-logs-btn"
          >
            <Trash2 className="w-3.5 h-3.5" />
            <span>Clear Logs</span>
          </button>
        )}
      </div>

      {logs.length === 0 ? (
        <div className="text-center py-12 bg-slate-950/40 rounded-lg border border-slate-800/50">
          <MessageSquare className="w-8 h-8 text-slate-600 mx-auto stroke-1" />
          <h4 className="text-slate-300 font-medium text-xs uppercase tracking-wider mt-3">No Signals Recorded Yet</h4>
          <p className="text-slate-500 text-xs mt-1 max-w-sm mx-auto">
            Use the Live Webhook trigger or the Alert Simulator tool above to run test confluences and generate logs.
          </p>
        </div>
      ) : (
        <div className="space-y-3">
          {logs.map(log => {
            const isSelected = selectedLogId === log.id;
            const isPassed = log.passedFilters;
            const side = log.payload?.utbot === "buy" ? "LONG" : "SHORT";
            const sideColor = side === "LONG" ? "text-emerald-400" : "text-rose-400";
            const scorePercent = Math.round((log.score / log.maxScore) * 100);

            return (
              <div
                key={log.id}
                className={`border rounded-xl transition-all overflow-hidden ${
                  isSelected
                    ? "bg-slate-950 border-cyan-500/30 shadow-md"
                    : isPassed
                    ? "bg-emerald-950/5 border-emerald-500/10 hover:border-slate-700 hover:bg-slate-900/30"
                    : "bg-rose-950/5 border-rose-500/10 hover:border-slate-700 hover:bg-slate-900/30"
                }`}
              >
                {/* Accordion trigger bar header */}
                <div
                  onClick={() => toggleDetails(log.id)}
                  className="p-3 md:p-4 flex flex-wrap items-center justify-between gap-3 cursor-pointer select-none"
                  id={`log-accordion-toggle-${log.id}`}
                >
                  <div className="flex items-center gap-3">
                    <span className={`text-[10px] uppercase font-bold px-2 py-1 rounded-md font-mono ${
                      isPassed
                        ? "bg-emerald-500/10 text-emerald-400 border border-emerald-500/25"
                        : "bg-rose-500/10 text-rose-400 border border-rose-500/25"
                    }`}>
                      {isPassed ? "PASSED MATCH" : "REJECTED"}
                    </span>
                    <div>
                      <div className="flex items-center gap-1.5">
                        <span className="font-semibold text-xs text-slate-200">{log.symbol}</span>
                        <span className="text-[10px] text-slate-500 font-mono">({log.timeframe})</span>
                        <span className={`text-[11px] font-bold ${sideColor}`}>{side}</span>
                      </div>
                      <span className="text-[10px] text-slate-500 font-mono block mt-0.5">
                        {new Date(log.timestamp).toLocaleTimeString()} · Price: ${log.price.toLocaleString()}
                      </span>
                    </div>
                  </div>

                  <div className="flex items-center gap-4">
                    <div className="text-right">
                      <span className="text-xs text-slate-400">Score:</span>
                      <span className={`font-mono font-bold text-xs ml-1 ${
                        isPassed ? "text-emerald-400" : "text-rose-400"
                      }`}>
                        {log.score}/{log.maxScore}
                      </span>
                    </div>

                    {isPassed && log.aiDecision && (
                      <div className="text-right hidden sm:block">
                        <span className="text-xs text-slate-400">AI Confidence:</span>
                        <span className="font-mono font-bold text-xs text-cyan-400 ml-1">
                          {log.aiDecision.confidence}%
                        </span>
                      </div>
                    )}

                    <div className="text-slate-400 hover:text-slate-200">
                      {isSelected ? <ChevronUp className="w-5 h-5" /> : <ChevronDown className="w-5 h-5" />}
                    </div>
                  </div>
                </div>

                {/* Log detailed breakdowns */}
                {isSelected && (
                  <div className="border-t border-slate-800 p-4 md:p-5 space-y-5 bg-slate-950/60 transition-all text-xs">
                    
                    {/* Filter results and scorecard */}
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                      {/* Left: Criteria scoring breaks */}
                      <div className="bg-slate-900 border border-slate-800/80 p-3.5 rounded-lg space-y-2">
                        <h4 className="text-[10px] font-bold text-cyan-400 tracking-wider uppercase border-b border-slate-800 pb-1.5">
                          Score Contribution Matrix
                        </h4>
                        <div className="space-y-1.5 pt-1">
                          {Object.entries(log.scoreBreakdown).map(([criteria, points]) => (
                            <div key={criteria} className="flex justify-between text-[11px] font-mono">
                              <span className="text-slate-400">{criteria}</span>
                              <span className={points > 0 ? "text-emerald-400 font-bold" : "text-slate-500"}>
                                +{points} pts
                              </span>
                            </div>
                          ))}
                        </div>
                      </div>

                      {/* Right: Block filters reasons mapping */}
                      <div className="bg-slate-900 border border-slate-800/80 p-3.5 rounded-lg space-y-2 flex flex-col justify-between">
                        <div>
                          <h4 className="text-[10px] font-bold text-cyan-400 tracking-wider uppercase border-b border-slate-800 pb-1.5">
                            Automated Filter Evaluation
                          </h4>
                          <div className="space-y-1.5 pt-2">
                            <div className="flex items-center justify-between text-[11px]">
                              <span className="text-slate-400">Low Volume Blocker:</span>
                              <span className={log.filterResults.lowVolume ? "text-rose-400 font-bold" : "text-emerald-400"}>
                                {log.filterResults.lowVolume ? "🔴 Trashed" : "🟢 Cleared"}
                              </span>
                            </div>
                            <div className="flex items-center justify-between text-[11px]">
                              <span className="text-slate-400">Hull Trend Alignment:</span>
                              <span className={log.filterResults.againstTrend ? "text-rose-400 font-bold" : "text-emerald-400"}>
                                {log.filterResults.againstTrend ? "🔴 Counter-trend blocker" : "🟢 Aligned"}
                              </span>
                            </div>
                            <div className="flex items-center justify-between text-[11px]">
                              <span className="text-slate-400">Market Structure Break:</span>
                              <span className={log.filterResults.noStructure ? "text-rose-400 font-bold" : "text-emerald-400"}>
                                {log.filterResults.noStructure ? "🔴 Missing structure" : "🟢 Structure Aligned"}
                              </span>
                            </div>
                            <div className="flex items-center justify-between text-[11px]">
                              <span className="text-slate-400">Inactive Liquidity hours:</span>
                              <span className={log.filterResults.deadSession ? "text-rose-400 font-bold" : "text-emerald-400"}>
                                {log.filterResults.deadSession ? "🔴 Dead session drop" : "🟢 Session Aligned"}
                              </span>
                            </div>
                          </div>
                        </div>

                        <div className="mt-2 text-[10px] text-slate-500 leading-relaxed border-t border-slate-800/50 pt-1.5">
                          Status: {isPassed ? "🟢 Confluences acceptable for dispatch." : "🔴 Suppressed to keep trading feed low-noise."}
                        </div>
                      </div>
                    </div>

                    {/* Multi-Timeframe Alignment Matrix if available */}
                    {log.multiTimeframe && log.multiTimeframe.length > 0 && (
                      <div className="bg-slate-900 border border-slate-800/80 p-3.5 rounded-lg space-y-2.5">
                        <h4 className="text-[10px] font-bold text-cyan-400 tracking-wider uppercase border-b border-slate-800 pb-1.5 flex items-center justify-between">
                          <span>Multi-Timeframe Correlation Grid</span>
                          <span className="text-[9px] text-slate-500 font-mono tracking-widest uppercase">1D to 1M Alignment check</span>
                        </h4>
                        <div className="grid grid-cols-4 sm:grid-cols-4 lg:grid-cols-8 gap-2">
                          {log.multiTimeframe.map((tf: any) => {
                            const isTrendBull = tf.trend === "bullish";
                            return (
                              <div key={tf.timeframe} className="p-2 bg-slate-950 rounded-lg border border-slate-800/60 text-center font-mono space-y-1">
                                <span className="text-[10px] font-black text-slate-300 block">{tf.timeframe}</span>
                                <div className={`text-[10px] font-extrabold py-0.5 rounded block ${isTrendBull ? "text-emerald-400 bg-emerald-950/40 border border-emerald-500/10" : "text-rose-400 bg-rose-950/40 border border-rose-500/10"}`}>
                                  {isTrendBull ? "▲ BULL" : "▼ BEAR"}
                                </div>
                                {tf.utbot !== "hold" && (
                                  <div className="text-[9px] text-cyan-400 font-extrabold bg-cyan-950/25 px-1 py-0.2 rounded border border-cyan-500/10 scale-95 truncate">
                                    ⚡ {tf.utbot.toUpperCase()}
                                  </div>
                                )}
                                {tf.structure !== "none" && (
                                  <div className="text-[8px] text-violet-400 font-medium scale-90 truncate">
                                    {tf.structure}
                                  </div>
                                )}
                              </div>
                            );
                          })}
                        </div>
                      </div>
                    )}

                    {/* Gemini AI Decision and Risk plans */}
                    {log.aiDecision && (
                      <div className="grid grid-cols-1 md:grid-cols-2 gap-4 border-t border-slate-800 pt-4">
                        {/* 1. Gemini analysis summary */}
                        <div className="bg-slate-900 border border-slate-800/80 p-3.5 rounded-lg space-y-2 relative overflow-hidden">
                          <div className="absolute top-0 right-0 w-24 h-24 bg-cyan-500/5 blur-2xl rounded-full" />
                          <h4 className="text-[10px] font-bold text-cyan-400 tracking-wider uppercase border-b border-slate-800 pb-1.5">
                            Gemini AI Evaluator Decision
                          </h4>
                          
                          <div className="flex items-center justify-between pb-1 pt-1">
                            <span className="text-slate-400">AI Verdict:</span>
                            <span className={`font-mono font-bold px-2 py-0.5 rounded ${
                              log.aiDecision.decision === "SEND" ? "bg-emerald-400/10 text-emerald-400" : "bg-rose-400/10 text-rose-400"
                            }`}>
                              {log.aiDecision.decision}
                            </span>
                          </div>

                          <div className="flex items-center justify-between pb-1">
                            <span className="text-slate-400">AI Confidence level:</span>
                            <span className="font-mono text-cyan-400 font-extrabold text-sm">{log.aiDecision.confidence}%</span>
                          </div>

                          <div className="pt-2 border-t border-slate-800/50 leading-relaxed text-slate-300">
                            <strong>Reason:</strong> {log.aiDecision.reason}
                          </div>
                        </div>

                        {/* 2. Calculated Risk targets */}
                        {log.tradePlan && (
                          <div className="bg-slate-900 border border-slate-800/80 p-3.5 rounded-lg space-y-2">
                            <h4 className="text-[10px] font-bold text-teal-400 tracking-wider uppercase border-b border-slate-800 pb-1.5">
                              Calculated ATR Risk Target limits
                            </h4>
                            
                            <div className="grid grid-cols-2 gap-x-4 gap-y-1 text-[11px] font-mono pt-1">
                              <div className="flex justify-between border-b border-slate-800 pb-1">
                                <span className="text-slate-500">Entry Limit:</span>
                                <span className="text-slate-200">${log.tradePlan.entry.toLocaleString()}</span>
                              </div>
                              <div className="flex justify-between border-b border-slate-800 pb-1">
                                <span className="text-slate-500 flex items-center gap-1">SL limit:</span>
                                <span className="text-rose-400 font-bold">${log.tradePlan.stopLoss.toLocaleString()}</span>
                              </div>
                              <div className="flex justify-between border-b border-slate-800 pb-1">
                                <span className="text-slate-500">TP Target 1:</span>
                                <span className="text-emerald-400">${log.tradePlan.takeProfit1.toLocaleString()}</span>
                              </div>
                              <div className="flex justify-between border-b border-slate-800 pb-1">
                                <span className="text-slate-500">TP Target 2:</span>
                                <span className="text-emerald-400">${log.tradePlan.takeProfit2.toLocaleString()}</span>
                              </div>
                              <div className="flex justify-between border-b border-slate-800 pb-1">
                                <span className="text-slate-500">TP Target 3:</span>
                                <span className="text-teal-400">${log.tradePlan.takeProfit3.toLocaleString()}</span>
                              </div>
                              <div className="flex justify-between border-b border-slate-800 pb-1">
                                <span className="text-slate-500">Target R:R:</span>
                                <span className="text-cyan-400">{log.tradePlan.riskRewardRatio}</span>
                              </div>
                            </div>
                          </div>
                        )}
                      </div>
                    )}

                    {/* Exact copyable telegram alert markdown formatting block */}
                    {isPassed && log.formattedAlert && (
                      <div className="border-t border-slate-800 pt-4 space-y-2">
                        <div className="flex justify-between items-center">
                          <h4 className="text-[10px] font-bold text-slate-400 tracking-wider uppercase flex items-center gap-1.5">
                            Formatted Telegram Notification
                          </h4>

                          <button
                            type="button"
                            onClick={() => copyAlertToClipboard(log, log.formattedAlert || "")}
                            className="px-2 py-1 bg-slate-900 border border-slate-850 hover:border-slate-755 hover:bg-slate-800 text-slate-300 hover:text-white rounded flex items-center gap-1 cursor-pointer select-none"
                          >
                            {copiedId === log.id ? (
                              <>
                                <Check className="w-3.5 h-3.5 text-emerald-400" />
                                <span className="text-[10px] text-emerald-400">Copied Alert</span>
                              </>
                            ) : (
                              <>
                                <Copy className="w-3.5 h-3.5" />
                                <span className="text-[10px]">Copy Alert Markdown</span>
                              </>
                            )}
                          </button>
                        </div>

                        <div className="relative">
                          <pre className="bg-slate-950 p-4 rounded-lg border border-slate-800/80 font-mono text-[11px] text-cyan-300/90 whitespace-pre-wrap select-all leading-relaxed">
                            {log.formattedAlert}
                          </pre>
                        </div>

                        {/* Telegram Delivery stats feedback */}
                        <div className="flex justify-between text-[11px] py-1 text-slate-500">
                          <span>Telegram Dispatch Trigger:</span>
                          <span className={log.telegramSent ? "text-emerald-400 font-bold" : "text-amber-400"}>
                            {log.telegramSent
                              ? "🚀 Sent successfully"
                              : log.telegramError
                              ? `⚠️ Failed: ${log.telegramError}`
                              : "⚠️ Switched Off / API Credentials Unregistered"}
                          </span>
                        </div>
                      </div>
                    )}
                  </div>
                )}
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}
