// Map ticker symbol back to valid TradingView exchange indexer notation
function getTradingViewSymbol(symbol: string): string {
  const cleanSym = symbol.toUpperCase().replace(".P", "");
  if (cleanSym.startsWith("XAU")) {
    return "OANDA:XAUUSD";
  }
  return `BINANCE:${cleanSym}`;
}

    const timer = setInterval(() => {
      setPairs(prev =>
        prev.map(p => {
          const delta = (Math.random() - 0.5) * p.price * 0.0016;
          if (Math.abs(delta) < 0.000001) return p;

          const newPrice = parseFloat((p.price + delta).toFixed(p.price > 1000 ? 1 : p.price > 10 ? 3 : p.price > 1 ? 4 : 8));
          const direction = newPrice > p.price ? "up" : "down";

          // Trigger a short flash state
          setLastUpdated(prevFlash => ({ ...prevFlash, [p.symbol]: direction }));
          setTimeout(() => {
            setLastUpdated(prevFlash => ({ ...prevFlash, [p.symbol]: null }));
          }, 800);

          // Randomly fluctuate volume and change values gently
          const currentChangeVal = parseFloat(p.change);
          const changeDelta = (Math.random() - 0.5) * 0.15;
          const newChange = (currentChangeVal + changeDelta).toFixed(2);
          const formattedChange = (parseFloat(newChange) >= 0 ? "+" : "") + newChange + "%";

          return { ...p, price: newPrice, change: formattedChange };
        })
      );
    }, 3000);

    return () => clearInterval(timer);

    const matchesSearch = p.symbol.toLowerCase().includes(searchQuery.toLowerCase());
    const matchesFilter = tabFilter === "all" || p.category === tabFilter;

          <div className="flex items-center gap-2">

              <Layers className="w-4 h-4 text-cyan-400" /> Active Market Ticker Feed

              placeholder="Search pairs..."

              All

              Futures

                tabFilter === "spot" ? "bg-slate-850 text-cyan-400 font-black" : "text-slate-400 hover:text-slate-200"
              }`}
            >
              Spot
            </button>
          </div>
        </div>
      </div>

      {/* Grid displaying the pairs beautifully */}

                        </span>
                      </div>
                    </div>

                    <div className="flex items-baseline justify-between pt-1">

                      <span className={`text-sm md:text-base font-black font-mono tracking-tight transition-colors duration-300 ${
                        flash === "up" ? "text-emerald-400 scale-[1.01]" : flash === "down" ? "text-rose-400 scale-[1.01]" : "text-slate-100"

          ● Stream Status: Normal
        </span>
      </div>

      {/* TradingView Live Chart Overlay modal */}
      {selectedPair && (
        <div className="fixed inset-0 bg-slate-950/80 backdrop-blur-sm z-50 flex items-center justify-center p-3 sm:p-5">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl w-full max-w-6xl h-[85vh] flex flex-col overflow-hidden shadow-2xl">
            {/* Modal Header */}
            <div className="p-4 sm:p-5 border-b border-slate-800 bg-slate-950/60 flex items-center justify-between shrink-0">
              <div className="flex items-center gap-3">
                <div className="bg-slate-900 p-2 rounded-xl border border-slate-850 flex items-center justify-center">
                  <TrendingUp className="w-5 h-5 text-cyan-400 animate-pulse" />
                </div>
                <div>
                  <div className="flex items-center gap-2">
                    <h3 className="text-sm sm:text-base font-black text-slate-100 uppercase tracking-tight">
                      {selectedPair.symbol} Interactive Trading Desk
                    </h3>
                    <span className={`text-[9px] font-mono uppercase tracking-wider px-2 py-0.5 rounded font-black border ${
                      selectedPair.category === "futures" 
                        ? "bg-indigo-950/50 text-indigo-300 border-indigo-500/20" 
                        : "bg-teal-950/50 text-teal-300 border-teal-500/20"
                    }`}>
                      {selectedPair.category}
                    </span>
                  </div>
                  <p className="text-[11px] text-slate-400 mt-0.5">
                    Live interactive TradingView charting layout matching default indicator configurations.
                  </p>
                </div>
              </div>

              {/* Action Toolbar */}
              <div className="flex items-center gap-3">
                {/* Timeframe selector */}

                  {[
                    { label: "5M", value: "5" },
                    { label: "15M", value: "15" },

                    { label: "4H", value: "240" },

                    <button
                      key={tf.value}
                      onClick={() => setChartInterval(tf.value)}
                      className={`px-3 py-1 text-[10px] uppercase font-bold rounded-md transition-all ${
                        chartInterval === tf.value ? "bg-slate-850 text-cyan-400 font-black shadow-sm" : "text-slate-500 hover:text-slate-200"

                    >
                      {tf.label}
                    </button>
                  ))}
                </div>

                {/* Close Button */}
                <button
                  onClick={() => setSelectedPair(null)}
                  className="p-2 bg-slate-950 border border-slate-855 rounded-lg text-slate-400 hover:text-slate-100 hover:border-slate-700 transition-all cursor-pointer shadow-sm"
                >
                  <X className="w-4 h-4" />
                </button>
              </div>
            </div>

            {/* Split viewport */}
            <div className="flex-1 flex flex-col lg:flex-row overflow-hidden min-h-0 bg-slate-950">
              {/* Mobile Timeframes */}
              <div className="flex sm:hidden bg-slate-950 border-b border-slate-850 p-2 justify-center gap-1.5 shrink-0">
                {[
                  { label: "5M", value: "5" },
                  { label: "15M", value: "15" },
                  { label: "1H", value: "60" },
                  { label: "4H", value: "240" },
                  { label: "1D", value: "D" }
                ].map(tf => (
                  <button
                    key={tf.value}
                    onClick={() => setChartInterval(tf.value)}
                    className={`px-3 py-1 text-[10px] font-black rounded transition-all ${
                      chartInterval === tf.value ? "bg-slate-850 text-cyan-400" : "text-slate-500 hover:text-slate-200"
                    }`}
                  >
                    {tf.label}
                  </button>
                ))}
              </div>

              {/* Viewport for TV Embed */}
              <div className="flex-1 h-full min-h-[300px] lg:min-h-0 relative">
                <iframe
                  id={`tradingview-${selectedPair.symbol}`}

                <div className="space-y-4">
                  <div className="space-y-1 pb-3 border-b border-slate-850">
                    <span className="text-[10px] uppercase font-bold text-slate-500 tracking-wider">Live Valuation</span>
                    <div className="flex items-baseline justify-between">
                      <span className="text-xl font-black font-mono text-white tracking-tight">
                        ${selectedPair.price.toLocaleString(undefined, { minimumFractionDigits: 2 })}
                      </span>
                      <span className={`text-xs font-bold font-mono py-0.5 px-2 rounded-md ${selectedPair.change.startsWith("+") ? "text-emerald-400 bg-emerald-950/30 border border-emerald-500/10" : "text-rose-400 bg-rose-950/30 border border-rose-500/10"}`}>
                        {selectedPair.change}
                      </span>
                    </div>
                  </div>

                  <div className="space-y-2.5">
                    <span className="text-[10px] uppercase font-bold text-slate-500 tracking-wider block">Indicator Checklist Confluences</span>

                    {[
                      { name: "UT Bot Indicator", state: "🟢 BULLISH BUY", desc: "Key breakout indicator threshold" },
                      { name: "Hull Suite Trend", state: "🟢 BULLISH", desc: "Fast moving average tracking" },
                      { name: "FVG Gap Suite", state: "🟢 UNFILLED GAP", desc: "Fair Value imbalances intact" },
                      { name: "Order Block (OB)", state: "🟢 DEMAND ZONE", desc: "Volume accumulation detected" },
