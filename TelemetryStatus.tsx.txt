import React from "react";
import { Check, ShieldCheck, Terminal, Copy, Radio } from "lucide-react";

interface TelemetryStatusProps {
  hasGeminiKey: boolean;
  webhookCount: number;
  pollingEnabled?: boolean;
}

export default function TelemetryStatus({ webhookCount, pollingEnabled }: TelemetryStatusProps) {
  const [copied, setCopied] = React.useState(false);
  const [showWebhookDetails, setShowWebhookDetails] = React.useState(false);
  const webhookUrl = `${window.location.origin}/api/webhook`;

  const handleCopy = () => {
    navigator.clipboard.writeText(webhookUrl);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className="bg-slate-900 border border-slate-800/85 rounded-xl p-4 shadow-md relative overflow-hidden" id="telemetry-status">
      <div className="absolute top-0 right-0 w-32 h-32 bg-cyan-500/5 blur-3xl rounded-full pointer-events-none" />
      
      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
        {/* Connection status and polling state summary */}
        <div className="flex items-center gap-3">
          <div className="relative border-r border-slate-800/80 pr-3.5 select-none">
            <span className="flex h-2.5 w-2.5">
              <span className="animate-ping absolute inline-flex h-2.5 w-2.5 rounded-full bg-emerald-400 opacity-75"></span>
              <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-emerald-500"></span>
            </span>
          </div>
          <div>
            <div className="flex items-center gap-2">
              <h3 className="font-bold text-xs tracking-wider text-slate-100 uppercase">
                {pollingEnabled ? "Headless Daemon Active" : "Alert Node Listening"}
              </h3>
              <span className="bg-slate-950 text-[8px] tracking-widest px-2 py-0.5 rounded-full text-cyan-400 font-mono border border-cyan-500/10 font-black uppercase">
                {pollingEnabled ? "Option A (100% Free)" : "Webhook Receiver"}
              </span>
            </div>
            <p className="text-[11px] text-slate-400 mt-0.5">
              {pollingEnabled 
                ? "Running round-the-clock server sessions to fetch asset prices & score setups instantly."
                : "Awaiting incoming Pine alert paylods from TradingView instances on port 3000."
              }
            </p>
          </div>
        </div>

        {/* Action Controls & Info Badges */}
        <div className="flex items-center gap-3 flex-wrap">
          {/* Quick toggle to reveal Webhook URL only if requested */}
          <button
            type="button"
            onClick={() => setShowWebhookDetails(!showWebhookDetails)}
            className="text-[10px] text-slate-400 hover:text-cyan-400 font-bold tracking-tight underline cursor-pointer"
          >
            {showWebhookDetails ? "Hide Webhook Config" : "Show Webhook URL"}
          </button>

          <div className="flex items-center gap-1.5 bg-slate-950/75 py-1 px-2.5 rounded border border-slate-850">
            <span className="text-[9px] font-bold text-slate-500 uppercase">Scanned Signals:</span>
            <span className="font-mono text-xs font-black text-cyan-400">{webhookCount}</span>
          </div>

          <div className="flex items-center gap-1.5 bg-slate-950/75 py-1 px-2.5 rounded border border-slate-850">
            <Terminal className="w-3 h-3 text-cyan-400 animate-pulse" />
            <span className="text-[9px] font-bold text-slate-500 uppercase">Engine:</span>
            <span className="font-mono text-[10px] font-black text-emerald-400">Gemini 3.5</span>
          </div>

          <div className="flex items-center gap-1.5 bg-slate-950/75 py-1 px-2.5 rounded border border-slate-850">
            <ShieldCheck className="w-3.5 h-3.5 text-emerald-400" />
            <span className="text-[10px] text-emerald-400 font-bold">Online</span>
          </div>
        </div>
      </div>

      {/* Animated toggle area for custom TradingView integration parameters */}
      {showWebhookDetails && (
        <div className="mt-3.5 pt-3.5 border-t border-slate-800/80 animate-fade-in space-y-2">
          <div className="flex items-center gap-1.5 text-[10px] font-bold text-slate-400 uppercase tracking-widest">
            <Radio className="w-3 h-3 text-cyan-400" /> Input Webhook Listener Endpoint
          </div>
          <div className="flex bg-slate-950 rounded-lg p-1.5 border border-slate-850 items-center justify-between max-w-2xl">
            <code className="text-xs text-slate-300 px-2 font-mono break-all overflow-hidden whitespace-nowrap text-ellipsis mr-2 select-all">
              {webhookUrl}
            </code>
            <button
              onClick={handleCopy}
              className="px-2.5 py-1 bg-slate-900 border border-slate-800 text-slate-300 rounded text-[10px] font-medium shrink-0 cursor-pointer hover:bg-slate-800 hover:text-white transition-all flex items-center gap-1.5"
            >
              {copied ? (
                <>
                  <Check className="w-3 h-3 text-emerald-400" />
                  <span className="text-emerald-400">Copied</span>
                </>
              ) : (
                <>
                  <Copy className="w-3 h-3" />
                  <span>Copy Link</span>
                </>
              )}
            </button>
          </div>
          <p className="text-[10px] text-slate-500 leading-normal">
            Note: Standard custom webhooks in TradingView strictly require an active <strong className="text-slate-400">Paid/Pro subscription</strong>. If you are on the free TradingView plan, keep the headless automated daemon enabled instead (Option A), which retrieves quote ticks entirely and cleanly for free!
          </p>
        </div>
      )}
    </div>
  );
}
